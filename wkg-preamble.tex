%---------------------------------------------------------------------
% Page-breaking behaviour
%   - never strand a single line at the top or bottom of a page
%   - \wkneed{N\baselineskip} makes a page break attractive when less
%     than N lines remain, so a heading cannot land at the page foot
%     with nothing under it (needspace's trick, inlined to avoid a
%     dependency)
%---------------------------------------------------------------------
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\makeatletter
\newcommand{\wkneed}[1]{%
  \begingroup
    \setlength{\dimen@}{#1}%
    \vskip\z@\@plus\dimen@
    \penalty-100
    \vskip\z@\@plus-\dimen@
  \endgroup}
\makeatother

% Role badges -- white on solid Azure, in two sizes.
%   \role     for the section heading (set \LARGE, so the badge sits at
%             \normalsize and reads as a tag, not a second title)
%   \roletoc  for the table of contents (set \normalsize, so the badge
%             drops to \scriptsize)
% Do not call these directly in a heading -- use \wksection or
% \wkrolesection below, which pick the right size per context.
% Robust: these are written to the .toc file, so they must not expand there.
% #1 size, #2 padding, #3 baseline raise, #4 text
\DeclareRobustCommand{\wkbadge}[4]{%
  {\setlength{\fboxsep}{#2}%
   \raisebox{#3}{%
     \colorbox{weknowazure}{\textcolor{white}{#1\bfseries #4}}}}}
\DeclareRobustCommand{\role}[1]{\texorpdfstring{%
  \hspace{0.45em}\wkbadge{\normalsize}{3.5pt}{-0.35ex}{#1}}{[#1]}}
\DeclareRobustCommand{\roletoc}[1]{\texorpdfstring{%
  \hspace{0.35em}\wkbadge{\scriptsize}{2pt}{-0.25ex}{#1}}{[#1]}}

%---------------------------------------------------------------------
% Section wrappers -- these set the running head and, where a role is
% given, place the badge at heading size in the body and at the smaller
% size in the table of contents.
%   \wksection{Final checklist}
%   \wkrolesection{Part 2 --- Install and lock the kiosk}{IT}
%---------------------------------------------------------------------
\newcommand{\wksection}[1]{%
  \renewcommand{\wksecmark}{#1}%
  \wkneed{6\baselineskip}%
  \section{#1}}
\newcommand{\wkrolesection}[2]{%
  \renewcommand{\wksecmark}{#1}%
  \wkneed{6\baselineskip}%
  \section[#1 \roletoc{#2}]{#1 \role{#2}}}
