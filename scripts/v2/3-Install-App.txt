<#
  EXPERIMENTAL v2, step 3 of 4 - not yet in the guide.
  Check the installer from step 2 is still the latest build in the feed, then
  install it per-user into the target account via a scheduled task and verify
  the installed version and identity file, then start Kato for 30s to prove it runs.
  Skips the install if Kiosk.exe already reports the latest version and the
  identity file already matches.
  Run as SYSTEM. SelfBooker need not be signed in: if it isn't, the install runs
  as SelfBooker in the background, with the password the machine keeps for its
  automatic sign-in, and the 30s start check is skipped.
  Next: 4-Lockdown.txt
#>

$TargetUser = 'SelfBooker'
$InstallerDir = "C:\Users\$TargetUser\Documents"
$KioskId = '{KIOSK_ID_FROM_COMMAND_CENTRE}'
$WorkstationId = $env:COMPUTERNAME # the machine label, e.g. WKG-POS-617
$ApiKey = '{API_KEY_FROM_COMMAND_CENTRE}'
$KioskFeedUrl = 'https://wkgdevstore.blob.core.windows.net/kiosk/' # electron-builder update feed
# From the N-central repository, pass any of these as command-line
# parameters instead: -KioskId <id> -ApiKey <key>

$ErrorActionPreference = 'Stop'
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$task = 'KioskInstall'
$start = Get-Date
$summary = [ordered]@{}
$script:warnings = @()
$script:current = ''

function Stamp { "[{0,5:n1}s]" -f ((Get-Date)-$start).TotalSeconds }
function Step($m) { $script:current = $m; Write-Host "$(Stamp) $m" }
function Ok($m) { Write-Host "$(Stamp) OK $m" -ForegroundColor Green }
function Warn($m) { $script:warnings += $m; Write-Host "$(Stamp) ! $m" -ForegroundColor Yellow }
function Show-Summary {
  Write-Host 'Summary:'
  foreach ($k in $summary.Keys) { Write-Host ("  {0,-12} {1}" -f $k, $summary[$k]) }
  foreach ($w in $script:warnings) { Write-Host "  Warning      $w" -ForegroundColor Yellow }
}

# version, file name, URL and SHA512 (hex) of the latest build in the feed
function Get-LatestBuild($feedUrl) {
  $feedUrl = $feedUrl.TrimEnd('/') + '/'
  Step "Reading ${feedUrl}latest.yml"
  $feed = (Invoke-WebRequest -Uri "${feedUrl}latest.yml" -UseBasicParsing).Content
  if ($feed -is [byte[]]) { $feed = [Text.Encoding]::UTF8.GetString($feed) }
  $yml = @{}
  foreach ($k in 'version', 'path', 'sha512') { # top-level keys only; the files: list is indented
    if ($feed -notmatch "(?m)^${k}:\s*'?([^'\r\n]+?)'?\s*$") { throw "latest.yml has no '$k'" }
    $yml[$k] = $Matches[1]
  }
  $build = [pscustomobject]@{
    Version = [version]$yml.version
    Name = $yml.path
    Url = $feedUrl + [uri]::EscapeDataString($yml.path)
    Sha512 = [BitConverter]::ToString([Convert]::FromBase64String($yml.sha512)).Replace('-', '')
  }
  Ok "Latest is v$($build.Version): $($build.Name)"
  $build
}

# Windows' LSA calls: grant and remove one user right for one account, as Local Security Policy does
# (every other account's rights are left alone), and read the stored automatic sign-in password.
# Compiled only when needed: it takes a few seconds
function Import-WkgLsa {
  if ('WkgLsa' -as [type]) { return }
  Add-Type -TypeDefinition @"
using System;
using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Security.Principal;
public static class WkgLsa {
  [StructLayout(LayoutKind.Sequential)] struct LsaString { public ushort Length, MaximumLength; public IntPtr Buffer; }
  [StructLayout(LayoutKind.Sequential)] struct LsaAttributes { public int Length; public IntPtr RootDirectory, ObjectName; public uint Attributes; public IntPtr SecurityDescriptor, SecurityQualityOfService; }
  [DllImport("advapi32.dll")] static extern uint LsaOpenPolicy(IntPtr system, ref LsaAttributes attributes, uint access, out IntPtr policy);
  [DllImport("advapi32.dll")] static extern uint LsaAddAccountRights(IntPtr policy, byte[] sid, LsaString[] rights, uint count);
  [DllImport("advapi32.dll")] static extern uint LsaRemoveAccountRights(IntPtr policy, byte[] sid, bool all, LsaString[] rights, uint count);
  [DllImport("advapi32.dll")] static extern uint LsaEnumerateAccountRights(IntPtr policy, byte[] sid, out IntPtr rights, out uint count);
  [DllImport("advapi32.dll")] static extern uint LsaRetrievePrivateData(IntPtr policy, LsaString[] key, out IntPtr data);
  [DllImport("advapi32.dll")] static extern uint LsaFreeMemory(IntPtr buffer);
  [DllImport("advapi32.dll")] static extern uint LsaClose(IntPtr policy);
  [DllImport("advapi32.dll")] static extern int LsaNtStatusToWinError(uint status);
  const uint NotFound = 0xC0000034; // the account holds no rights, or there is no such secret

  static void Check(uint status) { if (status != 0) throw new Win32Exception(LsaNtStatusToWinError(status)); }
  static byte[] Sid(string sid) { var s = new SecurityIdentifier(sid); var b = new byte[s.BinaryLength]; s.GetBinaryForm(b, 0); return b; }
  static IntPtr Open() { var a = new LsaAttributes(); IntPtr p; Check(LsaOpenPolicy(IntPtr.Zero, ref a, 0x00000815, out p)); return p; } // view, read secrets, create account, look up names
  static LsaString[] One(string right) {
    return new[] { new LsaString { Length = (ushort)(right.Length * 2), MaximumLength = (ushort)(right.Length * 2 + 2), Buffer = Marshal.StringToHGlobalUni(right) } };
  }

  public static bool Has(string sid, string right) {
    IntPtr p = Open(), list; uint n;
    try {
      uint status = LsaEnumerateAccountRights(p, Sid(sid), out list, out n);
      if (status == NotFound) return false;
      Check(status);
      try {
        int size = Marshal.SizeOf(typeof(LsaString));
        for (int i = 0; i < n; i++) {
          var r = (LsaString)Marshal.PtrToStructure(new IntPtr(list.ToInt64() + i * size), typeof(LsaString));
          if (Marshal.PtrToStringUni(r.Buffer, r.Length / 2) == right) return true;
        }
        return false;
      } finally { LsaFreeMemory(list); }
    } finally { LsaClose(p); }
  }
  public static void Add(string sid, string right) {
    IntPtr p = Open(); var r = One(right);
    try { Check(LsaAddAccountRights(p, Sid(sid), r, 1)); } finally { Marshal.FreeHGlobal(r[0].Buffer); LsaClose(p); }
  }
  public static string Secret(string name) {
    IntPtr p = Open(), data; var k = One(name);
    try {
      uint status = LsaRetrievePrivateData(p, k, out data);
      if (status == NotFound) return null;
      Check(status);
      try { var v = (LsaString)Marshal.PtrToStructure(data, typeof(LsaString)); return Marshal.PtrToStringUni(v.Buffer, v.Length / 2); }
      finally { LsaFreeMemory(data); }
    } finally { Marshal.FreeHGlobal(k[0].Buffer); LsaClose(p); }
  }
  public static void Remove(string sid, string right) {
    IntPtr p = Open(); var r = One(right);
    try { Check(LsaRemoveAccountRights(p, Sid(sid), false, r, 1)); } finally { Marshal.FreeHGlobal(r[0].Buffer); LsaClose(p); }
  }
}
"@
}

try {
  $names = 'TargetUser', 'InstallerDir', 'KioskId', 'ApiKey', 'KioskFeedUrl'
  for ($i = 0; $i -lt $args.Count; $i += 2) {
    $name = "$($args[$i])".TrimStart('-').TrimEnd(':')
    if ($name -notin $names) { throw "Unknown parameter '$($args[$i])'. Use: -$($names -join ', -')" }
    Set-Variable -Name $name -Value "$($args[$i + 1])"
  }

  Write-Host ""
  Write-Host "=== WKG Kiosk v2 step 3 - install the app on $WorkstationId ==="
  Write-Host ""

  Step 'Validating parameters'
  $checks = @{
    KioskId = @($KioskId, '^[0-9a-fA-F]{8}-([0-9a-fA-F]{4}-){3}[0-9a-fA-F]{12}$')
    WorkstationId = @($WorkstationId, '^[A-Za-z0-9][A-Za-z0-9-]{1,14}$')
    ApiKey = @($ApiKey, '^kcs_[A-Za-z0-9_-]{24,}$')
    KioskFeedUrl = @($KioskFeedUrl, '^https://\S+$')
  }
  foreach ($k in $checks.Keys) {
    $val, $pattern = $checks[$k]
    if ($val -cnotmatch $pattern) { throw "$k is malformed: '$val'" }
  }
  Ok 'Parameters well-formed'

  Step "Checking target user: $TargetUser"
  $user = Get-LocalUser -Name $TargetUser -ErrorAction SilentlyContinue
  if (-not $user) { throw "No local user '$TargetUser'" }
  # Not quser: with nobody signed in it writes an error, which stops the script under ErrorActionPreference Stop
  $signedIn = [bool](Get-Process -IncludeUserName | Where-Object UserName -like "*\$TargetUser")
  if ($signedIn) { Ok 'User present and logged on - installing in their session' }
  else { Ok "User present, not signed in - installing as $TargetUser in the background" }

  $build = Get-LatestBuild $KioskFeedUrl
  $exe = "C:\Users\$TargetUser\AppData\Local\Programs\Kiosk\Kiosk.exe"
  $json = "C:\Users\$TargetUser\AppData\Roaming\WKG Kiosk\kiosk-identity.json"
  # the version Kiosk.exe itself reports (its file properties), e.g. 1.1.1; $null if not installed
  function Get-InstalledVersion {
    if (-not (Test-Path -LiteralPath $exe)) { return $null }
    if ("$((Get-Item -LiteralPath $exe).VersionInfo.ProductVersion)" -match '^\d+\.\d+\.\d+') { [version]$Matches[0] }
  }
  # what in the identity file differs from this run's values; the API key is compared, never shown
  function Get-IdentityMismatch {
    if (-not (Test-Path -LiteralPath $json)) { return 'no identity file' }
    $cfg = Get-Content -LiteralPath $json -Raw | ConvertFrom-Json
    if ($cfg.kioskId -cne $KioskId) { "kioskId='$($cfg.kioskId)'" }
    if ($cfg.workstationId -cne $WorkstationId) { "workstationId='$($cfg.workstationId)'" }
    if ($null -ne $cfg.apiKey -and $cfg.apiKey -cne $ApiKey) { 'a different API key' }
  }

  Step 'Checking the installed app'
  $installed = Get-InstalledVersion
  $mismatch = @(Get-IdentityMismatch)
  $upToDate = $installed -eq $build.Version -and -not $mismatch
  if ($upToDate) { Ok "v$installed installed, the latest; identity matches - not reinstalling" }
  elseif (-not $installed) { Ok 'Not installed yet' }
  elseif ($installed -ne $build.Version) { Ok "v$installed installed - updating to v$($build.Version)" }
  else { Ok "v$installed installed, but the identity file has $($mismatch -join ', ') - reinstalling" }

  # Interactive: runs in the user's signed-in session, no password needed (the install and the start check)
  $principal = New-ScheduledTaskPrincipal -UserId $TargetUser -LogonType Interactive -RunLevel Limited
  if ($upToDate) {
    $summary['Installer'] = 'not needed - the latest version is already installed'
  } else {
    Step "Checking $InstallerDir has the latest build"
    $InstallerPath = Join-Path $InstallerDir $build.Name
    if (-not (Test-Path -LiteralPath $InstallerPath)) { throw "$($build.Name) not found in $InstallerDir - run step 2 first" }
    if ((Get-FileHash -LiteralPath $InstallerPath -Algorithm SHA512).Hash -ne $build.Sha512) {
      throw "$($build.Name) does not match the sha512 in latest.yml - run step 2 again"
    }
    $summary['Installer'] = "$($build.Name) (v$($build.Version), latest, sha512 verified)"
    Ok 'Installer is the latest build'

    Step 'Registering scheduled task'
    $argLine = "/S /KIOSKID=$KioskId /WORKSTATIONID=$WorkstationId /APIKEY=$ApiKey"
    Get-ScheduledTask -TaskName $task -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false
    $action = New-ScheduledTaskAction -Execute $InstallerPath -Argument $argLine
    $granted = $false
    try {
      if (-not $signedIn) {
        Import-WkgLsa
        # Without a session Windows needs the user's password: it refuses a no-password (S4U) task on these
        # kiosks, even to SYSTEM. Use the one the machine keeps for the user's automatic sign-in
        $winlogon = Get-ItemProperty -LiteralPath 'HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon'
        if ("$($winlogon.AutoAdminLogon)" -ne '1' -or $winlogon.DefaultUserName -ne $TargetUser) {
          throw "$TargetUser is not signed in and does not sign in automatically, so there is no stored password - sign in as $TargetUser, then run again"
        }
        $password = if ($winlogon.DefaultPassword) { $winlogon.DefaultPassword } else { [WkgLsa]::Secret('DefaultPassword') }
        if (-not $password) { throw "No stored sign-in password for $TargetUser - sign in as $TargetUser, then run again" }
        Ok "Using $TargetUser's stored sign-in password"

        # a password task also needs 'Log on as a batch job', which standard users lack. Give it to the
        # user for the install only; the finally below takes it back
        $sid = $user.SID.Value
        if (-not [WkgLsa]::Has($sid, 'SeBatchLogonRight')) { # already has it: leave it alone
          $granted = $true # set first: if the add half-works, the finally still takes it back
          [WkgLsa]::Add($sid, 'SeBatchLogonRight')
          if (-not [WkgLsa]::Has($sid, 'SeBatchLogonRight')) { throw "Could not give $TargetUser the 'Log on as a batch job' right" }
          $summary['Rights'] = "'Log on as a batch job' given to $TargetUser for the install"
          Ok "$TargetUser may log on as a batch job (until the install is done)"
        }
      }
      # the ScheduledTasks cmdlets ignore $ErrorActionPreference (it doesn't reach their module), so say Stop on each
      if ($signedIn) { $registered = Register-ScheduledTask -TaskName $task -Action $action -Principal $principal -Force -ErrorAction Stop }
      else { $registered = Register-ScheduledTask -TaskName $task -Action $action -User $TargetUser -Password $password -RunLevel Limited -Force -ErrorAction Stop }
      $registered | Start-ScheduledTask -ErrorAction Stop
      Ok 'Task registered and started'

      Step 'Waiting for installer to start (max 15s)'
      $by = (Get-Date).AddSeconds(15)
      while (($t = Get-ScheduledTask -TaskName $task).State -ne 'Running' -and (Get-Date) -lt $by) { Start-Sleep 1 }
      if ($t.State -ne 'Running') { # not running: fine only if it already ran and finished
        $result = ($t | Get-ScheduledTaskInfo).LastTaskResult
        if ($result -ne 0) { throw ("Task did not start. LastTaskResult 0x{0:X}" -f $result) }
      }
      Ok 'Installer running'

      Step 'Waiting for task to complete (max 120s)'
      $by = (Get-Date).AddSeconds(120)
      while (($t = Get-ScheduledTask -TaskName $task).State -eq 'Running' -and (Get-Date) -lt $by) { Start-Sleep 2 }
      if ($t.State -eq 'Running') { Warn 'Install task still running at timeout' }
      else { Ok 'Task finished' }

      Step 'Waiting for setup process to exit (max 60s)'
      $setupName = [IO.Path]::GetFileNameWithoutExtension($InstallerPath)
      $by = (Get-Date).AddSeconds(60)
      while ((Get-Process -Name $setupName -ErrorAction SilentlyContinue) -and (Get-Date) -lt $by) {
        Start-Sleep 2
      }
      if (Get-Process -Name $setupName -ErrorAction SilentlyContinue) {
        Warn 'Setup still alive (likely session 0, no desktop) - stopped it'
        Get-Process -Name $setupName -ErrorAction SilentlyContinue | Stop-Process -Force
      } else {
        Ok 'Setup exited cleanly'
      }
    }
    finally {
      $password = $null
      Get-ScheduledTask -TaskName $task -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false
      if ($granted) {
        try { [WkgLsa]::Remove($sid, 'SeBatchLogonRight') } catch {}
        $still = try { [WkgLsa]::Has($sid, 'SeBatchLogonRight') } catch { $true }
        if ($still) {
          Warn "$TargetUser may still have 'Log on as a batch job' - remove it in Local Security Policy > User Rights Assignment"
        } elseif ($summary['Rights']) { $summary['Rights'] += ', then taken back' }
      }
    }
  }

  Step 'Verifying install'
  if (-not (Test-Path -LiteralPath $exe)) { throw "Kiosk.exe missing at $exe" }
  $installed = Get-InstalledVersion
  if ($installed -ne $build.Version) { throw "Kiosk.exe reports v$installed after the install, expected v$($build.Version)" }
  Ok "Kiosk.exe is v$installed"

  if (-not (Test-Path -LiteralPath $json)) { throw "Not provisioned, no file at $json" }
  $mismatch = @(Get-IdentityMismatch)
  if ($mismatch) { throw "Config mismatch: $($mismatch -join '; ')" }
  $cfg = Get-Content -LiteralPath $json -Raw | ConvertFrom-Json
  $summary['App'] = "v$installed $(if ($upToDate) { 'already installed (not reinstalled)' } else { 'installed' }) for $TargetUser, identity matches $WorkstationId, printer $($cfg.printerDestination)"
  Ok 'Identity file matches'

  if (-not $signedIn) {
    # Kato needs a desktop to show on; step 4's restart signs SelfBooker in and starts it
    $summary['Launch'] = "not checked - $TargetUser was not signed in; the restart in step 4 starts Kato"
  } else {
    Step "Starting Kato as $TargetUser (30s)"
    # a bad config (e.g. wrong API endpoints) installs fine but Kato closes straight away
    function Get-Kato { @(Get-Process -Name Kiosk -IncludeUserName -ErrorAction SilentlyContinue | Where-Object UserName -like "*\$TargetUser") }
    # already running = already locked down; closing it would restart the machine (Shell Launcher action 3)
    $kato = Get-Kato
    $wasRunning = [bool]$kato
    if (-not $wasRunning) {
      Register-ScheduledTask -TaskName $task -Action (New-ScheduledTaskAction -Execute $exe) -Principal $principal -Force -ErrorAction Stop | Start-ScheduledTask -ErrorAction Stop
      Start-Sleep 30
      $kato = Get-Kato
      $kato | Stop-Process -Force # step 4's lockdown starts it for real
      Get-ScheduledTask -TaskName $task -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false
    }
    if (-not $kato) { throw "Kato closed within 30s of starting - check its config and $json" }
    $summary['Launch'] = if ($wasRunning) { "Kato already running as $TargetUser (locked down), left running" } else { "Kato ran for 30s as $TargetUser, then was closed" }
    Ok 'Kato kept running'
  }

  Write-Host ""
  Write-Host ("=== SUCCESS in {0:n1}s (v{1}) ===" -f ((Get-Date)-$start).TotalSeconds, $build.Version) -ForegroundColor Green
  Show-Summary
  Write-Host "Next: 4-Lockdown.txt"
  Write-Host ""
  exit 0
}
catch {
  Write-Host ""
  Write-Host ("=== FAILED after {0:n1}s ===" -f ((Get-Date)-$start).TotalSeconds) -ForegroundColor Red
  Write-Host $_.Exception.Message -ForegroundColor Red
  $summary['Failed at'] = $script:current
  Show-Summary
  Write-Host ""
  Get-ScheduledTask -TaskName $task -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false
  exit 1
}
