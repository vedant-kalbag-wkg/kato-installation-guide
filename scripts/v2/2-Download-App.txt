<#
  EXPERIMENTAL v2, step 2 of 4 - not yet in the guide.
  Download the latest Kato app installer named in the update feed's latest.yml
  into the installer folder, and check it against the feed's sha512.
  Run as SYSTEM. Next: 3-Install-App.txt
#>

$InstallerDir = 'C:\Users\SelfBooker\Documents'
$KioskFeedUrl = 'https://wkgdevstore.blob.core.windows.net/kiosk/' # electron-builder update feed
$RequireVersion = $null # e.g. '1.1.1' to refuse anything else; $null accepts whatever latest.yml names

$ErrorActionPreference = 'Stop'
$ProgressPreference = 'SilentlyContinue' # the progress bar slows large downloads badly on PowerShell 5.1
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$start = Get-Date
$summary = [ordered]@{}
$script:warnings = @()
$script:current = ''

function Stamp { "[{0,5:n1}s]" -f ((Get-Date)-$start).TotalSeconds }
function Step($m) { $script:current = $m; Write-Host "$(Stamp) $m" }
function Ok($m) { Write-Host "$(Stamp) OK $m" -ForegroundColor Green }
function Warn($m) { $script:warnings += $m; Write-Host "$(Stamp) ! $m" -ForegroundColor Yellow }
function Show-Summary {
  Write-Host 'Summary:'
  foreach ($k in $summary.Keys) { Write-Host ("  {0,-12} {1}" -f $k, $summary[$k]) }
  foreach ($w in $script:warnings) { Write-Host "  Warning      $w" -ForegroundColor Yellow }
}

# version, file name, URL and SHA512 (hex) of the latest build in the feed
function Get-LatestBuild($feedUrl) {
  $feedUrl = $feedUrl.TrimEnd('/') + '/'
  Step "Reading ${feedUrl}latest.yml"
  $feed = (Invoke-WebRequest -Uri "${feedUrl}latest.yml" -UseBasicParsing).Content
  if ($feed -is [byte[]]) { $feed = [Text.Encoding]::UTF8.GetString($feed) }
  $yml = @{}
  foreach ($k in 'version', 'path', 'sha512') { # top-level keys only; the files: list is indented
    if ($feed -notmatch "(?m)^${k}:\s*'?([^'\r\n]+?)'?\s*$") { throw "latest.yml has no '$k'" }
    $yml[$k] = $Matches[1]
  }
  $build = [pscustomobject]@{
    Version = [version]$yml.version
    Name = $yml.path
    Url = $feedUrl + [uri]::EscapeDataString($yml.path)
    Sha512 = [BitConverter]::ToString([Convert]::FromBase64String($yml.sha512)).Replace('-', '')
  }
  Ok "Latest is v$($build.Version): $($build.Name)"
  $build
}

try {
  $names = 'InstallerDir', 'KioskFeedUrl', 'RequireVersion'
  for ($i = 0; $i -lt $args.Count; $i += 2) {
    $name = "$($args[$i])".TrimStart('-').TrimEnd(':')
    if ($name -notin $names) { throw "Unknown parameter '$($args[$i])'. Use: -$($names -join ', -')" }
    Set-Variable -Name $name -Value "$($args[$i + 1])"
  }

  Write-Host ""
  Write-Host "=== WKG Kiosk v2 step 2 - download the app on $env:COMPUTERNAME ==="
  Write-Host ""

  if ($KioskFeedUrl -notmatch '^https://\S+$') { throw "KioskFeedUrl is not an https:// link: '$KioskFeedUrl'" }
  $build = Get-LatestBuild $KioskFeedUrl
  if ($RequireVersion -and $build.Version -ne [version]$RequireVersion) {
    throw "Expected v$RequireVersion, feed has v$($build.Version)"
  }

  $target = Join-Path $InstallerDir $build.Name
  if ((Test-Path -LiteralPath $target) -and (Get-FileHash -LiteralPath $target -Algorithm SHA512).Hash -eq $build.Sha512) {
    Ok "$($build.Name) already downloaded, sha512 matches latest.yml - not downloading again"
    $summary['Installer'] = "$target (v$($build.Version), already present, sha512 verified)"
  } else {
    Step 'Downloading Kato app installer'
    New-Item -ItemType Directory -Force -Path $InstallerDir | Out-Null
    # a part-download left by an earlier run can still be held open (e.g. by antivirus), so never reuse one
    Get-ChildItem -LiteralPath $InstallerDir -Filter "$($build.Name).*download" -File | Remove-Item -Force -ErrorAction SilentlyContinue
    $partial = "$target.$([guid]::NewGuid().ToString('N').Substring(0, 8)).download"
    Invoke-WebRequest -Uri $build.Url -OutFile $partial -UseBasicParsing
    if ((Get-FileHash -LiteralPath $partial -Algorithm SHA512).Hash -ne $build.Sha512) {
      Remove-Item -LiteralPath $partial -Force
      throw 'Downloaded installer does not match the sha512 in latest.yml'
    }
    Move-Item -LiteralPath $partial -Destination $target -Force
    $file = Get-Item -LiteralPath $target
    Ok ("Downloaded {0} ({1:n0} KB), sha512 matches latest.yml" -f $build.Name, ($file.Length / 1KB))

    # only the x.y.z part: a label such as 1.2.0-beta.1 or a fourth part (1.2.0.0) is not a different version
    $fileVer = "$($file.VersionInfo.ProductVersion)"
    if ($fileVer -match '^\d+\.\d+\.\d+' -and [version]$Matches[0] -ne $build.Version) {
      Warn "latest.yml says v$($build.Version) but file metadata says v$fileVer"
    }
    $summary['Installer'] = "$target (v$($build.Version), sha512 verified)"
  }

  Write-Host ""
  Write-Host ("=== SUCCESS in {0:n1}s ===" -f ((Get-Date)-$start).TotalSeconds) -ForegroundColor Green
  Show-Summary
  Write-Host "Next: 3-Install-App.txt"
  Write-Host ""
  exit 0
}
catch {
  Write-Host ""
  Write-Host ("=== FAILED after {0:n1}s ===" -f ((Get-Date)-$start).TotalSeconds) -ForegroundColor Red
  Write-Host $_.Exception.Message -ForegroundColor Red
  $summary['Failed at'] = $script:current
  Show-Summary
  Write-Host ""
  exit 1
}
